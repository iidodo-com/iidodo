' ===========================================================================
'  ThisWorkbook に貼り付けるコード
'  Alt+F11 → 左のプロジェクトで「ThisWorkbook」をダブルクリック → 下の内容をすべて貼り付け
'  （mod予算書ブック.bas を先にインポートしておく）
' ===========================================================================
Option Explicit

' 手入力セルの黄色は画面表示用。印刷の前に黄色表示を止め、印刷が済んだら元に戻す。
Private Sub Workbook_BeforePrint(Cancel As Boolean)
    Call SuspendInputColor
End Sub

' 印刷後の最初の操作で黄色表示を戻す
Private Sub Workbook_SheetSelectionChange(ByVal Sh As Object, ByVal Target As Range)
    Call ResumeInputColor
End Sub

Private Sub Workbook_SheetActivate(ByVal Sh As Object)
    Call ResumeInputColor
End Sub

' 黄色表示が止まったまま保存・終了しないようにする
Private Sub Workbook_BeforeSave(ByVal SaveAsUI As Boolean, Cancel As Boolean)
    Call ResumeInputColor
End Sub

Private Sub Workbook_BeforeClose(Cancel As Boolean)
    Call ResumeInputColor
    Call DisableEnterKey
End Sub

' Enterキーで次の手入力セルへ移動する機能。このブックがアクティブな間だけキー割り当てを変える
Private Sub Workbook_Open()
    Call EnableEnterKey
End Sub

Private Sub Workbook_Activate()
    Call EnableEnterKey
End Sub

Private Sub Workbook_Deactivate()
    Call DisableEnterKey
End Sub

Private Sub Workbook_SheetChange(ByVal Sh As Object, ByVal Target As Range)
    Call OnInputCellChanged(Target)
End Sub
