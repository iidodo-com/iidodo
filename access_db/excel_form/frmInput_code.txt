Private Function AddTxt(nm As String, y As Single, w As Single, Optional h As Single = 18, Optional multi As Boolean = False) As MSForms.TextBox

        t.MultiLine = True: t.EnterKeyBehavior = True: t.ScrollBars = fmScrollBarsVertical

End Function

' 1列を上から詰めてコンボに入れる。lastRow=0 なら使用範囲の最終行まで。
Private Sub FillList(c As MSForms.ComboBox, ws As Worksheet, col As Long, firstRow As Long, lastRow As Long)
    Dim r As Long, n As Long, v As Variant
    c.Clear
    If lastRow = 0 Then n = ws.Cells(ws.Rows.Count, col).End(xlUp).Row Else n = lastRow
    For r = firstRow To n
        v = ws.Cells(r, col).Value
        If Len(CStr(v)) > 0 Then c.AddItem CStr(v)
    Next
End Sub

' リスト シートの「キー列(keyCol)=key」の行の、次列の値をコンボに入れる
Private Sub FillByKey(c As MSForms.ComboBox, keyCol As Long, key As String)
    Dim ws As Worksheet, n As Long, r As Long, v As Variant
    Set ws = ThisWorkbook.Worksheets("リスト")
    c.Clear
    n = ws.Cells(ws.Rows.Count, keyCol).End(xlUp).Row
    If n < 2 Then Exit Sub
    v = ws.Range(ws.Cells(2, keyCol), ws.Cells(n, keyCol + 1)).Value
    For r = 1 To UBound(v, 1)
        If CStr(v(r, 1)) = key Then c.AddItem CStr(v(r, 2))
    Next
End Sub

' ---------- 連動 ----------
Private Sub cmb1_AfterUpdate()
    cmb2.Clear: cmb3.Clear: cmbCat.Clear: cmbProb.Clear
    FillByKey cmb2, 3, cmb1.Value
End Sub

Private Sub cmb2_AfterUpdate()
    cmb3.Clear: cmbCat.Clear: cmbProb.Clear
    FillByKey cmb3, 6, cmb1.Value & "|" & cmb2.Value
    If cmb3.ListCount = 0 Then
        ' 項目(3)が無い事務は、そのまま区分へ
        FillByKey cmbCat, 9, cmb1.Value & "|" & cmb2.Value & "|"
    End If
End Sub

Private Sub cmb3_AfterUpdate()
    cmbCat.Clear: cmbProb.Clear
    FillByKey cmbCat, 9, cmb1.Value & "|" & cmb2.Value & "|" & cmb3.Value
End Sub

Private Sub cmbCat_AfterUpdate()
    cmbProb.Clear
    FillByKey cmbProb, 12, cmb1.Value & "|" & cmb2.Value & "|" & cmb3.Value & "|" & cmbCat.Value
End Sub

' ---------- 登録 ----------
Private Function NextRow(ws As Worksheet) As Long
    Dim r As Long
    r = ws.Cells(ws.Rows.Count, "K").End(xlUp).Row + 1
    If r < 2 Then r = 2
    If r > ROW_LAST Then NextRow = 0 Else NextRow = r
End Function

Private Sub btnReg_Click()
    Dim ws As Worksheet, r As Long
    lblMsg.Caption = ""
    If Not IsNumeric(txtYear.Value) Or Len(txtYear.Value) = 0 Then MsgBox "監査年度を数字で入力してください。", vbExclamation: Exit Sub
    If Len(Trim(txtDept.Value)) = 0 Then MsgBox "対象局課名を入力してください。", vbExclamation: Exit Sub
    If Len(txtDate.Value) > 0 Then
        If Not IsDate(txtDate.Value) Then MsgBox "実地監査年月日は 2026/6/4 の形式で入力してください。", vbExclamation: Exit Sub
    End If
    If Len(cmbProb.Value) = 0 Then MsgBox "問題点を選択してください。", vbExclamation: Exit Sub
    If Left(cmbProb.Value, 5) = "（その他）" And Len(Trim(txtNaiyo.Value)) = 0 Then
        MsgBox "「（その他）」を選んだ場合は、内容(要約)を入力してください。", vbExclamation: Exit Sub
    End If

    Set ws = ThisWorkbook.Worksheets("入力")
    r = NextRow(ws)
    If r = 0 Then MsgBox "入力欄の最終行に達しました。入力シートの数式行を下へコピーして行を増やしてください。", vbExclamation: Exit Sub

    ws.Range("B" & r).Value = CLng(txtYear.Value)
    If Len(txtDate.Value) > 0 Then ws.Range("C" & r).Value = CDate(txtDate.Value)
    ws.Range("D" & r).Value = Trim(txtDept.Value)
    ws.Range("E" & r).Value = cmbGroup.Value
    ws.Range("F" & r).Value = txtTanto.Value
    ws.Range("G" & r).Value = cmb1.Value
    ws.Range("H" & r).Value = cmb2.Value
    ws.Range("I" & r).Value = cmb3.Value
    ws.Range("J" & r).Value = cmbCat.Value
    ws.Range("K" & r).Value = cmbProb.Value
    ws.Range("M" & r).Value = txtNaiyo.Value
    ws.Range("N" & r).Value = txtShosai.Value
    ws.Range("O" & r).Value = cmbStatus.Value
    ws.Range("P" & r).Value = cmbBiko.Value
    ws.Range("Q" & r).Value = txtKonkyo.Value
    ws.Range("R" & r).Value = txtKw.Value
    ws.Range("S" & r).Value = IIf(chkShuchi.Value, "要", "不要")

    lblMsg.Caption = "登録しました(入力シート " & r & " 行目)。続けて入力できます。"
    ClearDetail
End Sub

Private Sub ClearDetail()
    cmb1.ListIndex = -1
    cmb2.Clear: cmb3.Clear: cmbCat.Clear: cmbProb.Clear
    txtNaiyo.Value = "": txtShosai.Value = ""
    cmbStatus.ListIndex = -1: cmbBiko.ListIndex = -1
    txtKonkyo.Value = "": txtKw.Value = ""
    chkShuchi.Value = False
End Sub

Private Sub btnClr_Click()
    ClearDetail
    lblMsg.Caption = ""
End Sub

Private Sub btnClose_Click()
    Unload Me
End Sub
