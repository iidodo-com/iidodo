Private Function AddLbl(p As Object, cap As String, x As Single, y As Single, w As Single, Optional req As Boolean = False) As MSForms.Label

' 1列を上から詰めてコンボに入れる。lastRow=0 なら最初の空白セルの手前まで。
Private Sub FillList(c As MSForms.ComboBox, ws As Worksheet, col As Long, firstRow As Long, lastRow As Long)
    Dim r As Long, v As Variant
    busy = True
    c.Clear
    r = firstRow
    Do
        If lastRow > 0 And r > lastRow Then Exit Do
        v = ws.Cells(r, col).Value
        If Len(CStr(v)) = 0 Then
            If lastRow = 0 Then Exit Do
        Else
            c.AddItem CStr(v)
        End If
        r = r + 1
        If r > 5000 Then Exit Do
    Loop
    busy = False
End Sub

' 空白・全角空白を除いて比較用に整える
Private Function Norm(ByVal s As String) As String
    Norm = Replace(Replace(s, ChrW(12288), ""), " ", "")
End Function

' リスト シートの「キー列(keyCol)=key」の行の、次列の値をコンボに入れる。入れた件数を返す
Private Function FillByKey(c As MSForms.ComboBox, keyCol As Long, key As String) As Long
    Dim ws As Worksheet, n As Long, r As Long, v As Variant, k As String, cnt As Long
    Set ws = ThisWorkbook.Worksheets("リスト")
    c.Clear
    n = ws.Cells(ws.Rows.Count, keyCol).End(xlUp).Row
    If n < 2 Then Exit Function
    v = ws.Range(ws.Cells(2, keyCol), ws.Cells(n, keyCol + 1)).Value
    k = Norm(key)
    For r = 1 To UBound(v, 1)
        If Norm(CStr(v(r, 1))) = k Then
            c.AddItem CStr(v(r, 2)): cnt = cnt + 1
        End If
    Next
    FillByKey = cnt
End Function

Private Sub ShowCount(nm As String, n As Long)
    If n = 0 Then
        lblMsg.Caption = nm & ": 候補がありません。"
    Else
        lblMsg.Caption = nm & ": " & n & " 件の候補があります。"
    End If
End Sub

' ---------- 連動(選択が変わったら下位の候補を作り直す) ----------
Private Sub FillCat()
    Dim n As Long
    n = FillByKey(cmbCat, 9, cmb1.Value & "|" & cmb2.Value & "|" & cmb3.Value)
    ShowCount "区分", n
End Sub

Private Sub cmb1_Change()
    If busy Then Exit Sub
    busy = True
    cmb2.Clear: cmb3.Clear: cmbCat.Clear: cmbProb.Clear
    If cmb1.ListIndex >= 0 Then ShowCount "項目(2)", FillByKey(cmb2, 3, cmb1.Value)
    busy = False
End Sub

Private Sub cmb2_Change()
    If busy Then Exit Sub
    busy = True
    cmb3.Clear: cmbCat.Clear: cmbProb.Clear
    If cmb2.ListIndex >= 0 Then
        ShowCount "項目(3)", FillByKey(cmb3, 6, cmb1.Value & "|" & cmb2.Value)
        If cmb3.ListCount = 0 Then
            FillCat      ' 項目(3)が無い事務は、そのまま区分へ
            lblMsg.Caption = "この項目には項目(3)がありません。区分を選んでください。"
        End If
    End If
    busy = False
End Sub

Private Sub cmb3_Change()
    If busy Then Exit Sub
    busy = True
    cmbCat.Clear: cmbProb.Clear
    If cmb3.ListIndex >= 0 Then FillCat
    busy = False
End Sub

Private Sub cmbCat_Change()
    If busy Then Exit Sub
    busy = True
    cmbProb.Clear
    If cmbCat.ListIndex >= 0 Then
        ShowCount "問題点", FillByKey(cmbProb, 12, cmb1.Value & "|" & cmb2.Value & "|" & cmb3.Value & "|" & cmbCat.Value)
    End If
    busy = False
End Sub

' ---------- 登録 ----------
Private Function NextRow(ws As Worksheet) As Long
    Dim r As Long
    r = ws.Cells(ws.Rows.Count, "K").End(xlUp).Row + 1
    If r < 2 Then r = 2
    If r > ROW_LAST Then NextRow = 0 Else NextRow = r
End Function

Private Sub btnReg_Click()
    Dim ws As Worksheet, r As Long
    lblMsg.Caption = ""
    If Len(txtYear.Value) = 0 Or Not IsNumeric(txtYear.Value) Then MsgBox "監査年度を数字で入力してください。", vbExclamation: Exit Sub
    If Len(Trim(txtDept.Value)) = 0 Then MsgBox "対象局課名を入力してください。", vbExclamation: Exit Sub
    If Len(txtDate.Value) > 0 Then
        If Not IsDate(txtDate.Value) Then MsgBox "実地監査年月日は 2026/6/4 の形式で入力してください。", vbExclamation: Exit Sub
    End If
    If Len(cmbProb.Value) = 0 Then MsgBox "問題点を選択してください。", vbExclamation: Exit Sub
    If Left(cmbProb.Value, 5) = "（その他）" And Len(Trim(txtNaiyo.Value)) = 0 Then
        MsgBox "「（その他）」を選んだ場合は、内容(要約)を入力してください。", vbExclamation: Exit Sub
    End If

    Set ws = ThisWorkbook.Worksheets("入力")
    r = NextRow(ws)
    If r = 0 Then MsgBox "入力欄の最終行に達しました。入力シートの数式行を下へコピーして行を増やしてください。", vbExclamation: Exit Sub

    ws.Range("B" & r).Value = CLng(txtYear.Value)
    If Len(txtDate.Value) > 0 Then ws.Range("C" & r).Value = CDate(txtDate.Value)
    ws.Range("D" & r).Value = Trim(txtDept.Value)
    ws.Range("E" & r).Value = cmbGroup.Value
    ws.Range("F" & r).Value = txtTanto.Value
    ws.Range("G" & r).Value = cmb1.Value
    ws.Range("H" & r).Value = cmb2.Value
    ws.Range("I" & r).Value = cmb3.Value
    ws.Range("J" & r).Value = cmbCat.Value
    ws.Range("K" & r).Value = cmbProb.Value
    ws.Range("M" & r).Value = txtNaiyo.Value
    ws.Range("N" & r).Value = txtShosai.Value
    ws.Range("O" & r).Value = cmbStatus.Value
    ws.Range("P" & r).Value = cmbBiko.Value
    ws.Range("Q" & r).Value = txtKonkyo.Value
    ws.Range("R" & r).Value = txtKw.Value
    ws.Range("S" & r).Value = IIf(chkShuchi.Value, "要", "不要")

    ClearDetail
    lblMsg.Caption = "登録しました(入力シート " & r & " 行目)。続けて入力できます。"
End Sub

Private Sub ClearDetail()
    busy = True
    cmb1.ListIndex = -1
    cmb2.Clear: cmb3.Clear: cmbCat.Clear: cmbProb.Clear
    txtNaiyo.Value = "": txtShosai.Value = ""
    cmbStatus.ListIndex = -1: cmbBiko.ListIndex = -1
    txtKonkyo.Value = "": txtKw.Value = ""
    chkShuchi.Value = False
    busy = False
End Sub

Private Sub btnClr_Click()
    ClearDetail
    lblMsg.Caption = ""
End Sub

Private Sub btnClose_Click()
    Unload Me
End Sub
