' ---------- リストの読み込み ----------
' 1列を上から詰めてコンボに入れる。lastRow=0 なら最初の空白セルの手前まで。
Private Sub FillList(c As MSForms.ComboBox, ws As Worksheet, col As Long, firstRow As Long, lastRow As Long)
    Dim r As Long, v As Variant
    busy = True
    c.Clear
    r = firstRow
    Do
        If lastRow > 0 And r > lastRow Then Exit Do
        v = ws.Cells(r, col).Value
        If Len(CStr(v)) = 0 Then
            If lastRow = 0 Then Exit Do
        Else
            c.AddItem CStr(v)
        End If
        r = r + 1
        If r > 5000 Then Exit Do
    Loop
    busy = False
End Sub

' 空白・全角空白を除いて比較用に整える
Private Function Norm(ByVal s As String) As String
    Norm = Replace(Replace(s, ChrW(12288), ""), " ", "")
End Function

' リストシートの「キー列(keyCol)=key」の行の、次列の値をコンボに入れる。入れた件数を返す
Private Function FillByKey(c As MSForms.ComboBox, keyCol As Long, key As String) As Long
    Dim ws As Worksheet, n As Long, r As Long, v As Variant, k As String, cnt As Long
    Set ws = ThisWorkbook.Worksheets("リスト")
    c.Clear
    n = ws.Cells(ws.Rows.Count, keyCol).End(xlUp).Row
    If n < 2 Then Exit Function
    v = ws.Range(ws.Cells(2, keyCol), ws.Cells(n, keyCol + 1)).Value
    k = Norm(key)
    For r = 1 To UBound(v, 1)
        If Norm(CStr(v(r, 1))) = k Then
            c.AddItem CStr(v(r, 2)): cnt = cnt + 1
        End If
    Next
    FillByKey = cnt
End Function

Private Sub ShowCount(nm As String, n As Long)
    If n = 0 Then lblMsg.Caption = nm & ": 候補がありません。" Else lblMsg.Caption = nm & ": " & n & " 件の候補があります。"
End Sub

' コンボの項目を文字で選ぶ(無ければ未選択)
Private Sub SelectItem(c As MSForms.ComboBox, ByVal t As String)
    Dim i As Long
    c.ListIndex = -1
    If Len(t) = 0 Then Exit Sub
    For i = 0 To c.ListCount - 1
        If Norm(CStr(c.List(i))) = Norm(t) Then c.ListIndex = i: Exit Sub
    Next
End Sub

' ---------- 行の読込(修正) ----------
Private Function IsRowEmpty(ws As Worksheet, r As Long) As Boolean
    IsRowEmpty = (Application.WorksheetFunction.CountA(ws.Range("B" & r & ":N" & r)) = 0)
End Function

Private Sub btnLoad_Click()
    Dim n As Long, r As Long, ws As Worksheet
    lblMsg.Caption = ""
    If Len(txtNo.Value) = 0 Or Not IsNumeric(txtNo.Value) Then MsgBox "No.を数字で入力してください。", vbExclamation: Exit Sub
    n = CLng(txtNo.Value): r = n + 1
    If r < 2 Or r > ROW_LAST Then MsgBox "No.は 1〜" & (ROW_LAST - 1) & " の範囲で入力してください。", vbExclamation: Exit Sub
    Set ws = ThisWorkbook.Worksheets("入力")
    If IsRowEmpty(ws, r) Then MsgBox "No." & n & " の行にはデータがありません。", vbExclamation: Exit Sub
    LoadRow ws, r
    mRow = r
    ShowMode
    lblMsg.Caption = "No." & n & " を読み込みました。直したら「修正を保存」を押してください。"
End Sub

Private Sub LoadRow(ws As Worksheet, r As Long)
    busy = True
    txtYear.Value = CStr(ws.Range("B" & r).Value)
    If IsDate(ws.Range("C" & r).Value) Then txtDate.Value = Format(ws.Range("C" & r).Value, "yyyy/mm/dd") Else txtDate.Value = ""
    txtTanto.Value = CStr(ws.Range("I" & r).Value)
    SelectItem cmbGroup, CStr(ws.Range("H" & r).Value)
    ' 所属
    SelectItem cmbD, CStr(ws.Range("D" & r).Value)
    FillByKey cmbE, KC_E, CStr(cmbD.Value)
    SelectItem cmbE, CStr(ws.Range("E" & r).Value)
    FillByKey cmbF, KC_F, cmbD.Value & "|" & cmbE.Value
    SelectItem cmbF, CStr(ws.Range("F" & r).Value)
    FillByKey cmbG, KC_G, cmbD.Value & "|" & cmbE.Value & "|" & cmbF.Value
    SelectItem cmbG, CStr(ws.Range("G" & r).Value)
    ' 分類
    SelectItem cmb1, CStr(ws.Range("J" & r).Value)
    FillByKey cmb2, KC_2, CStr(cmb1.Value)
    SelectItem cmb2, CStr(ws.Range("K" & r).Value)
    FillByKey cmb3, KC_3, cmb1.Value & "|" & cmb2.Value
    SelectItem cmb3, CStr(ws.Range("L" & r).Value)
    FillByKey cmbCat, KC_C, cmb1.Value & "|" & cmb2.Value & "|" & cmb3.Value
    SelectItem cmbCat, CStr(ws.Range("M" & r).Value)
    FillByKey cmbProb, KC_P, cmb1.Value & "|" & cmb2.Value & "|" & cmb3.Value & "|" & cmbCat.Value
    SelectItem cmbProb, CStr(ws.Range("N" & r).Value)
    ' 内容など
    txtNaiyo.Value = CStr(ws.Range("P" & r).Value)
    txtShosai.Value = CStr(ws.Range("Q" & r).Value)
    SelectItem cmbStatus, CStr(ws.Range("R" & r).Value)
    SelectItem cmbBiko, CStr(ws.Range("S" & r).Value)
    txtKonkyo.Value = CStr(ws.Range("T" & r).Value)
    txtKw.Value = CStr(ws.Range("U" & r).Value)
    chkShuchi.Value = (CStr(ws.Range("V" & r).Value) = "要")
    busy = False
End Sub

' ---------- 登録・修正の保存 ----------
Private Function NextRow(ws As Worksheet) As Long
    Dim v As Variant, i As Long, j As Long, blank As Boolean
    v = ws.Range("B2:N" & ROW_LAST).Value
    For i = 1 To UBound(v, 1)
        blank = True
        For j = 1 To UBound(v, 2)
            If IsError(v(i, j)) Then
                blank = False: Exit For
            ElseIf Len(CStr(v(i, j))) > 0 Then
                blank = False: Exit For
            End If
        Next
        If blank Then NextRow = i + 1: Exit Function
    Next
    NextRow = 0
End Function

    If Len(txtYear.Value) = 0 Or Not IsNumeric(txtYear.Value) Then MsgBox "監査年度を数字で入力してください。", vbExclamation: Exit Sub
    If Len(cmbD.Value) = 0 Or Len(cmbE.Value) = 0 Or Len(cmbG.Value) = 0 Then MsgBox "所属(部局・局・課)を選択してください。", vbExclamation: Exit Sub
    If Len(txtDate.Value) > 0 Then
        If Not IsDate(txtDate.Value) Then MsgBox "実地監査年月日は 2026/6/4 の形式で入力してください。", vbExclamation: Exit Sub
    End If

    If Left(cmbProb.Value, 5) = "（その他）" And Len(Trim(txtNaiyo.Value)) = 0 Then
        MsgBox "「（その他）」を選んだ場合は、内容(要約)を入力してください。", vbExclamation: Exit Sub
    End If

        r = mRow
    Else

    End If

    ws.Range("B" & r).Value = CLng(txtYear.Value)
    If Len(txtDate.Value) > 0 Then ws.Range("C" & r).Value = CDate(txtDate.Value) Else ws.Range("C" & r).ClearContents
    ws.Range("D" & r).Value = cmbD.Value
    ws.Range("E" & r).Value = cmbE.Value
    ws.Range("F" & r).Value = cmbF.Value
    ws.Range("G" & r).Value = cmbG.Value
    ws.Range("H" & r).Value = cmbGroup.Value
    ws.Range("I" & r).Value = txtTanto.Value
    ws.Range("J" & r).Value = cmb1.Value
    ws.Range("K" & r).Value = cmb2.Value
    ws.Range("L" & r).Value = cmb3.Value
    ws.Range("M" & r).Value = cmbCat.Value
    ws.Range("N" & r).Value = cmbProb.Value
    ws.Range("P" & r).Value = txtNaiyo.Value
    ws.Range("Q" & r).Value = txtShosai.Value
    ws.Range("R" & r).Value = cmbStatus.Value

    End If

End Sub

Private Sub ClearDetail()
    busy = True
    cmb1.ListIndex = -1
    cmb2.Clear: cmb3.Clear: cmbCat.Clear: cmbProb.Clear
    txtNaiyo.Value = "": txtShosai.Value = ""
    cmbStatus.ListIndex = -1: cmbBiko.ListIndex = -1
    txtKonkyo.Value = "": txtKw.Value = ""
    chkShuchi.Value = False
    txtNo.Value = ""
    busy = False
End Sub

Private Sub btnNew_Click()
    mRow = 0
    ShowMode
    ClearDetail
    lblMsg.Caption = "新規入力にしました。"
End Sub

Private Sub btnDel_Click()
    Dim ws As Worksheet
    If mRow = 0 Then MsgBox "削除するには、先にNo.を入力して「読込」してください。", vbExclamation: Exit Sub
    If MsgBox("No." & (mRow - 1) & " を削除しますか?" & vbCrLf & "(その行の入力内容を空にします)", vbYesNo + vbQuestion, "削除の確認") <> vbYes Then Exit Sub
    Set ws = ThisWorkbook.Worksheets("入力")
    ws.Range("B" & mRow & ":N" & mRow).ClearContents
    ws.Range("P" & mRow & ":V" & mRow).ClearContents
    lblMsg.Caption = "No." & (mRow - 1) & " を削除しました。"
    mRow = 0
    ShowMode
    ClearDetail
End Sub

Private Sub btnClose_Click()
    Unload Me
End Sub
